Работа с файловой системой

Модуль path

 
							
								const path = require('node:path');

								const notes = '/users/joe/notes.txt';

								path.dirname(notes); // /users/joe
								path.basename(notes); // notes.txt
								path.extname(notes); // .txt

								// Объединение  нескольких частей пути
								const name = 'joe';
								path.join('/', 'users', name, 'notes.txt'); // '/users/joe/notes.txt'

								// Получение абсолютного пути из относительного
								path.resolve('joe.txt'); // '/Users/joe/joe.txt' если запустить из домашней директории
								path.resolve('tmp', 'joe.txt'); // '/Users/joe/tmp/joe.txt'


								// Вычисление пути, когда есть относительные спецификаторы
								path.normalize('/users/joe/..//test.txt'); // '/users/test.txt'

Стримы

Стрим – это абстракция, с помощью которой можно обрабатывать данные небольшими частями

Пример. Читаем целиком в память

 
							
								const fs = require('node:fs');
								const path = require('node:path');
								const server = require('node:http').createServer();

								server.on('request', (req, res) => {
									const src = fs.readFile(path.resolve(__dirname, './ClickerDemo.key'), (err, data) => {
										res.end(data);
									});	
								})

								server.listen(3000);

Пример. Возвращаем по частям

 
							
								const fs = require('node:fs');
								const path = require('node:path');
								const server = require('node:http').createServer();

								server.on('request', (req, res) => {
									const src = fs.createReadStream(path.resolve(__dirname, './ClickerDemo.key'));
									src.pipe(res);
								})

								server.listen(3000);

Пример. Все-равно все в памяти

 
							
								const fs = require('node:fs');
								const path = require('node:path');
								const server = require('node:http').createServer();

								server.on('request', (req, res) => {
									let buffer = '';

									const src = fs.createReadStream(path.resolve(__dirname, './ClickerDemo.key'));

									src.on('data', data => {
										buffer += data;
									});

									src.on('end', data => {
										res.end(res);
									})
								})

								server.listen(3000);

Модульная система

Организация nodejs проекта

В NodeJS каждый отдельный файл рассматривается как модуль

Одновременно есть две системы модулей

  • CommonJS
  • ECMAScript

CommonJS

circle.js 
							
								const { PI } = Math;

								exports.area = (r) => PI * r ** 2;

								exports.circumference = (r) => 2 * PI * r;
main.js 
							
								const circle = require("./circle.js");
								console.log(`The area of a circle of radius 4 is ${circle.area(4)}`);

Переменные внутри модуля доступны только в этом модуле

Для того, чтобы вернуть что-то из модуля используется module.exports

							
								const { PI } = Math;

								const area = (r) => PI * r ** 2;

								const circumference = (r) => 2 * PI * r;

								module.exports = { area, circumference };

exports - ссылка на объект module.exports

В module.exports можно положить не только объект, но и класс или функцию

Модуль выполняется один раз при первом require()

То есть объект, возвращаемый из require('foo-module') всегда один и тот же

В NodeJS есть встроенные модули, которые поставляются бинарниками. Обращаться к ним можно через префикс node:, например

							
								const fs = require("node:fs");

Есть core модули, которые можно подключать и без префикса node:, их список можно получить:

							
								const builtin = require("node:module").builtinModules;
resolution → loading → wrapping → evaluation → caching

3 основных типа модулей

  • Встроенные core
  • Написанные в процессе разработки
  • Установленные через npm
							
	require(X) from module at path Y
	1. If X is a core module,
		a. return the core module
		b. STOP
	2. If X begins with '/'
		a. set Y to be the file system root
	3. If X begins with './' or '/' or '../'
		a. LOAD_AS_FILE(Y + X)
		b. LOAD_AS_DIRECTORY(Y + X)
		c. THROW "not found"
	4. LOAD_NODE_MODULES(X, dirname(Y))
	5. THROW "not found"

						

	LOAD_AS_FILE(X)
	1. If X is a file, 
		load X as its file extension format. STOP
	2. If X.js is a file, 
		load X.js as JavaScript text. STOP
	3. If X.json is a file, 
		parse X.json to a JavaScript Object. STOP
	4. If X.node is a file, 
		load X.node as binary addon. STOP

						

	LOAD_AS_DIRECTORY(X)
	1. If X/package.json is a file,
		a. Parse X/package.json, and look for "main" field.
		b. If "main" is a falsy value, GOTO 2.
		c. let M = X + (json main field)
		d. LOAD_AS_FILE(M)
		e. LOAD_INDEX(M)
		f. LOAD_INDEX(X) DEPRECATED
		g. THROW "not found"
	2. LOAD_INDEX(X)

	LOAD_INDEX(X)
	1. If X/index.js is a file, 
		load X/index.js as JavaScript text. STOP
	2. If X/index.json is a file, 
		parse X/index.json to a JavaScript object. STOP
	3. If X/index.node is a file, 
		load X/index.node as binary addon. STOP

						

	NODE_MODULES_PATHS(START)
	1. let PARTS = path split(START)
	2. let I = count of PARTS - 1
	3. let DIRS = []
	4. while I >= 0,
		a. if PARTS[I] = "node_modules" CONTINUE
		b. DIR = path join(PARTS[0 .. I] + "node_modules")
		c. DIRS = DIR + DIRS
		d. let I = I - 1
	5. return DIRS + GLOBAL_FOLDERS

Перед выполнением кода в модуле NodeJS обернет его в функцию:

Это дает изоляцию переменным и пробрасывает необходимые переменные

							
								(function(exports, require, module, __filename, __dirname) {
									// Module code actually lives in here
								});

__dirname - название директории, в которой находится модуль

__filename - полный абсолютный путь до текущего файла

Event Loop - программная конструкция, которая ожидает прибытия и производит рассылку событий или сообщений в программе

libuv - библиотека на C, создающая абстракцию над операциям ввода/вывода на разных ОС и добавляющая Event Loop

https://blog.insiderattack.net/event-loop-and-the-big-picture-nodejs-event-loop-part-1-1cb67a182810 
							
								int uv_run(uv_loop_t* loop, uv_run_mode mode) {
									int timeout;
									int r;
									int can_sleep;
								  
									r = uv__loop_alive(loop);
									if (!r)
									  uv__update_time(loop);
								  
									/* Maintain backwards compatibility by processing timers before entering the
									 * while loop for UV_RUN_DEFAULT. Otherwise timers only need to be executed
									 * once, which should be done after polling in order to maintain proper
									 * execution order of the conceptual event loop. */
									if (mode == UV_RUN_DEFAULT && r != 0 && loop->stop_flag == 0) {
									  uv__update_time(loop);
									  uv__run_timers(loop);
									}
								  
									while (r != 0 && loop->stop_flag == 0) {
									  can_sleep =
										  uv__queue_empty(&loop->pending_queue) &&
										  uv__queue_empty(&loop->idle_handles);
								  
									  uv__run_pending(loop); // Запуск колбэков всех системных операций ввода/вывода
									  uv__run_idle(loop);
									  uv__run_prepare(loop);
								  
									  timeout = 0;
									  if ((mode == UV_RUN_ONCE && can_sleep) || mode == UV_RUN_DEFAULT)
										timeout = uv__backend_timeout(loop);
								  
									  uv__metrics_inc_loop_count(loop);
								  
									  uv__io_poll(loop, timeout); // Ожидание завершения ввода вывода
								  
									  /* Process immediate callbacks (e.g. write_cb) a small fixed number of
									   * times to avoid loop starvation.*/
									  for (r = 0; r < 8 && !uv__queue_empty(&loop->pending_queue); r++)
										uv__run_pending(loop);
								  
									  /* Run one final update on the provider_idle_time in case uv__io_poll
									   * returned because the timeout expired, but no events were received. This
									   * call will be ignored if the provider_entry_time was either never set (if
									   * the timeout == 0) or was already updated b/c an event was received.
									   */
									  uv__metrics_update_idle_time(loop);
								  
									  uv__run_check(loop); // Запуск setImmediate
									  uv__run_closing_handles(loop); // Коллбэки на close события
								  
									  uv__update_time(loop);
									  uv__run_timers(loop); // Запуск заэкспайренных таймеров
								  
									  r = uv__loop_alive(loop);
									  if (mode == UV_RUN_ONCE || mode == UV_RUN_NOWAIT)
										break;
									}
								  
									/* The if statement lets gcc compile it to a conditional store. Avoids
									 * dirtying a cache line.
									 */
									if (loop->stop_flag != 0)
									  loop->stop_flag = 0;
								  
									return r;
								  }
