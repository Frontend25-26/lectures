именованный блок кода для реиспользования

Синтаксис создания функций

							
								// Function Declaration
								function showMessage() {
									console.log('Добро пожаловать');
								}
								showMessage();


								// Function Expression
								let showMessage = function() {
									console.log('Добро пожаловать');
								}
								showMessage();

								// Arrow function
								let showMessage = () => {
									console.log('Добро пожаловать');
								}
								showMessage();

								// new Function
								let showMessage = new Function("console.log('1')");
								showMessage();

Имена функций

  • функция - это действие
  • одна функция - одно действие
  • имя должно точно описывать действие
  • часто используют глагольные префиксы get, check, map, calc, ...

Область видимости

1. Локальные переменные

							
								function showMessage() {
									// переменная видна только внутри функции
									let message = 'Добро пожаловать';

									console.log(message);
								}

								showMessage(); // 'Добро пожаловать'

								console.log(message); // ошибка

2. Внешние переменные

							
								let message = 'Добро пожаловать';

								function showMessage() {
									console.log(message);
								}

								showMessage(); // 'Добро пожаловать'

								message = 'Welcome'

								showMessage(); // 'Welcome'

Перекрытие переменных

							
								let message = 'Добро пожаловать';

								function showMessage() {
									let message = 'Welcome';
									console.log(message);
								}

								showMessage(); // 'Welcome'

								message = 'Изменили текст';
								// значение локальной переменной не изменено
								showMessage(); // 'Welcome'

3. Глобальные переменные

  • Переменные, объявленные снаружи всех функций
  • Переменные видны для любой функции

Аргументы

Параметры, передаваемые в функцию, называются аргументами

							
								function showMessage(text) {
									console.log(text);
								}

								showMessage('Наше сообщение');

Можно передавать любое кол-во аргументов

							
								function sum(value1, value2) {
									return value1 + value2;
								}

								sum(1,2,3,4);

Некоторые из аргументов могут быть необязательными

							
								function showMessage(value1, value2) {
									if (value1) {
										console.log(value1);
									}

									console.log(value2);
								}

								showMessage(undefined, 'текст');

Если аргументов становится много

Остаточные параметры

Постановка проблемы

							
								function sum(value1, value2) {
									return value1 + value2;
								}

								sum(1,2,3,4);

Перепишем решение

							
								// args - массив всех из всех аргументов
								function sum(...args) {
									let result = 0;

									for (let i = 0; i < args.length; i++) {
										result += args[i];
									}

									return result;
								}

								sum(1,2,3,4); // 10

spread & rest синтаксис

Примеры использования

							
								const numbers = [1,2,3,4,5];
								const [one, two, ...other] = numbers; // spread syntax

								const obj = { name: 'Name', age: 20 };
								const newObj = { ...obj, age: 21 }; // spread syntax

								const fn = (...args) => { // rest parameters
									// своя логика
								}
								fn(...numbers); // spread syntax

Объединять аргументы не обязательно сначала

							
								function sum(arg1, arg2, ...args) {
									let result = arg1 + arg2;

									for (let i = 0; i < args.length; i++) {
										result += args[i];
									}

									return result;
								}

								sum(1,2,3,4); // 10

2й способ решения - arguments

							
								function sum() {
									let result = 0;

									for (let i = 0; i < arguments.length; i++) {
										result += arguments[i];
									}

									return result;
								}

								sum(1,2,3,4); // 10

Для стрелочной функции arguments не существуют

							
								const sum = () => {
									let result = 0;

									for (let i = 0; i < arguments.length; i++) {
										result += arguments[i];
									}

									return result;
								}

								sum(1,2,3,4); // ошибка

Как исправить? - использовать "...args" вместо "arguments"

Обратная задача

							
								let data = [1, 2, 3, 4];

									let sum = 0;

									for (let i = 0; i < args.length; i++) {
										sum += args[i];

Замыкание

это функция, которая запоминает значения переменных из своей области видимости, где была создана, и может получать к ним доступ, даже если она вызывается в другом месте, где эти переменные не видны.

Пример

							
								function showMessage() {
									let message = 'Добро пожаловать';

									console.log(message);
								}

								showMessage(); // 'Добро пожаловать'

Лексическое окружение (LexicalEnvironment)


Объект состоит из частей:

  • Environment Record
  • Ссылка на внешнее лексическое окружение
							
								const lexicalEnvironment = {
									environmentRecord: {...},
									outer: ...
								}
							
						
							
								function showMessage() {
									let message = 'Добро пожаловать';

									console.log(message);
								}

								showMessage(); // 'Добро пожаловать'
							
						
							
								const lexicalEnvironment1 = {
									environmentRecord: {
										showMessage: function
									},
									outer: null
								}
							
						
							
								const lexicalEnvironment2 = {
									environmentRecord: {
										message: >uninitialized<
									},
									outer: lexicalEnvironment1
								}
  • Все функции помнят, где они были созданы с помощью скрытого свойства [[Environment]]
  • Ссылка на [[Environment]] устанавливается один раз и навсегда при создании функции
  • Лексическое окружение удаляется из памяти вместе со всеми переменными после завершения вызова функции

Пример функции, возвращающую другую функцию

							
								function makeWelcome() {
									let message = 'Добро пожаловать';

									return function() {
										console.log(message);
									}
								}

								let welcome = makeWelcome();
								welcome(); // 'Добро пожаловать'

								// или
								makeWelcome()(); // 'Добро пожаловать'

Пример создания счетчика

							
								function makeCounter() {
									let count = 0;

									return function() {
										return count++;
									};
								}

								let counter = makeCounter();
								counter(); // 0
								counter(); // 1
								counter(); // 2

								let counter2 = makeCounter();
								counter2(); // 0

Где еще встречается лексическое окружение?

  • функции
  • условия
  • циклы
  • блоки кода

Примеры лексического окружения. Условие

							
								if (true) {
									let message = 'text';

									console.log(message); // 'text'
								}

								console.log(message); // Ошибка, нет такой переменной

Примеры лексического окружения. Циклы

							
								for (let i = 0; i < 10; i++) {
								  	// У каждой итерации цикла своё лексическое окружение
								}

								console.log(i); // Ошибка, нет такой переменной

Примеры лексического окружения. Блоки кода

							
								{
									let message = 'text';

									console.log(message); // 'text'
								}

								{
									let message = 'text2';

									console.log(message); // 'text2'
								}

								console.log(message); // Ошибка, нет такой переменной

Итого

  • Любая функция является замыканием
  • Все функции содержат скрытое свойство [[Environment]]
  • Один вызов функции - одно лексическое окружение
  • Правила сборки мусора такие же как и для объектов

this

  • Ключевое слово
  • Указывает на текущий контекст
  • Нельзя перезаписать

Значение this зависит от

  1. Окружения
  2. Как оказались в функции
  3. Режим работы интерпретатора

1. This. Окружение

							
								console.log('Hello world');

								// window - глобальный объект в браузере

								this.console.log('Hello world');

								this === window; // true

								// global - глобальный объект в node.js
								global.console.log('Hello world');

								// работает и в браузере и в nodejs
								globalThis.console.log('Hello world');

Чистая функция

Функция называется чистой, если она удовлетворяет 2 условиям

  1. Каждый раз функция возвращает одинаковый результат, когда она вызывается с тем же набором аргументов
  2. Отсутствуют побочные эффекты

Пример нечистой функции №1

							
								function getDayAfterTomorrow() {
									const oneDay = 24 * 60 * 60 * 1000;

									return new Date(Date.now() + oneDay)
								}
								getDayAfterTomorrow();

Как исправить?

								
									function getDayAfterTomorrow(currentTime) {
										const oneDay = 24 * 60 * 60 * 1000;

										return new Date(currentTime + oneDay)
									}

									const currentTime = Date.now()
									getDayAfterTomorrow(currentTime);
									getDayAfterTomorrow(currentTime);

Пример нечистой функции №2

							
								let time = Date.now();

								function getDayAfterTomorrow() {
									const oneDay = 24 * 60 * 60 * 1000;

									return new Date(time + oneDay)
								}

Пример нечистой функции №3

							
								function sum(a, b) {
									console.log('123');

									return a + b;
								}

Примеры побочных эффектов

  • http вызовы (fetch/ajax)
  • изменение входных данных
  • изменение файловой структуры
  • DOM запросы
  • console.log

Ошибка: Maximum call stack size exceeded

Исправляем пример

							
								const infiniteCounter = (num) => {

								}

									factorial(num - 1);
								}
